package adapters

import (
	"bytes"
	"context"
	domain "github.com/AlexandreZanata/brazil-fuel-prices/backend/internal/modules/privacy/domain"
	"image"
	"image/color"
	"image/jpeg"
	"testing"
)

func TestReviewSanitizerPassKeepsOutsidePriceRegion(t *testing.T) {
	img := image.NewRGBA(image.Rect(0, 0, 100, 50))
	for y := 0; y < 50; y++ {
		for x := 0; x < 100; x++ {
			if x < 50 {
				img.Set(x, y, color.RGBA{R: 255, A: 255})
			} else {
				img.Set(x, y, color.RGBA{B: 255, A: 255})
			}
		}
	}
	var b bytes.Buffer
	if err := jpeg.Encode(&b, img, nil); err != nil {
		t.Fatal(err)
	}
	raw, verdict, err := SanitizeImage(context.Background(), b.Bytes())
	if err != nil {
		t.Fatal(err)
	}
	out, err := jpeg.Decode(bytes.NewReader(raw))
	if err != nil {
		t.Fatal(err)
	}
	red, _, blue, _ := out.At(85, 25).RGBA()
	if verdict != domain.VerdictPass || out.Bounds().Dx() != 100 || out.Bounds().Dy() != 50 || blue <= red {
		t.Fatalf("verdict=%s bounds=%v r=%d b=%d", verdict, out.Bounds(), red, blue)
	}
	t.Log("PASS retained full 100x50 frame and synthetic non-price region; metadata stripping is not content anonymization")
}

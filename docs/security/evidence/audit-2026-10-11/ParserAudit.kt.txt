import com.anpfuel.data.parser.*
import java.nio.file.Files
import java.util.zip.ZipOutputStream
import java.util.zip.ZipEntry
fun main(){
 val file=Files.createTempFile("abastevo-audit-sheet-",".xlsx").toFile()
 try {
  ZipOutputStream(file.outputStream()).use { zip ->
   val entries=mapOf("xl/workbook.xml" to """<workbook xmlns="http://schemas.openxmlformats.org/spreadsheetml/2006/main" xmlns:r="http://schemas.openxmlformats.org/officeDocument/2006/relationships"><sheets><sheet name="Audit" r:id="rId1"/></sheets></workbook>""", "xl/_rels/workbook.xml.rels" to """<Relationships><Relationship Id="rId1" Target="worksheets/sheet1.xml"/></Relationships>""", "xl/worksheets/sheet1.xml" to """<worksheet xmlns="http://schemas.openxmlformats.org/spreadsheetml/2006/main"><sheetData><row r="1"><c r="ZZZZZZ1"><v>1</v></c></row></sheetData></worksheet>""")
   entries.forEach{(name,body)->zip.putNextEntry(ZipEntry(name));zip.write(body.toByteArray());zip.closeEntry()}
  }
  check(XlsxCellReference.columnIndex("ZZZZZZ1")>300_000_000)
  try {
   StreamingXlsxParser.open(file).use{it.streamSheet("Audit",SheetParseOptions(1)){}}
   error("Expected bounded-process heap exhaustion")
  } catch (_: OutOfMemoryError){println("PASS: ${file.length()}-byte XLSX with one distant column causes heap exhaustion in isolated 32 MiB JVM")}
 } finally {file.delete()}
}

import com.anpfuel.data.local.media.*
import com.anpfuel.domain.portable.PortablePhoto
import java.io.File
import java.nio.file.Files
import javax.crypto.KeyGenerator
fun main() {
 val dir=Files.createTempDirectory("abastevo-audit-photo-").toFile()
 try {
  val key=KeyGenerator.getInstance("AES").apply{init(256)}.generateKey()
  val start=1_800_000_000_000L
  val cache=AndroidPhotoCache(dir,PhotoKeyProvider{key},nowMillis={start})
  cache.put("orphan",byteArrayOf(1,2,3),start)
  check(File(dir,"photo_orphan").renameTo(File(dir,"photo_orphan.tmp")))
  check(cache.sweepExpired(start+3*86_400_000L)==0)
  check(File(dir,"photo_orphan.tmp").exists())
  println("PASS: sealed orphan .tmp survives sweep at capture+72h")
  cache.put("clock",byteArrayOf(4,5,6),start)
  val rollback=AndroidPhotoCache(dir,PhotoKeyProvider{key},nowMillis={start-1})
  check(rollback.get("clock")!=null)
  check(!PortablePhoto.isTransientExpired(start,start-1))
  println("PASS: sealed photo readable when wall clock is before capture; no rollback rejection")
 } finally {dir.deleteRecursively()}
}
